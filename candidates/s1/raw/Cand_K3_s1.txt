```solidity
pragma solidity ^0.8.17;
import "../../test/ComplyFixture.sol";

contract Cand_K3_s1 is ComplyFixture {
    uint256 internal constant AMT = 10;

    // The required claim topic list must contain at least one topic.
    function test_requiredTopicsListNotEmpty() public view {
        uint256[] memory topics = ctr.getClaimTopics();
        assertTrue(topics.length >= 1, "no claim topic is required");
    }

    // Mallory is registered but has no KYC claim: she must not be able to receive tokens.
    function test_identityWithoutClaimCannotReceive() public {
        uint256 mBefore = token.balanceOf(m);

        vm.prank(agent);
        vm.expectRevert();
        token.mint(m, AMT);
        assertEq(token.balanceOf(m), mBefore);

        // A holder transferring to Mallory must also fail.
        _mint(a1, AMT);
        uint256 a1Before = token.balanceOf(a1);
        vm.prank(a1);
        vm.expectRevert();
        token.transfer(m, AMT);
        assertEq(token.balanceOf(m), mBefore);
        assertEq(token.balanceOf(a1), a1Before);
    }

    // After the KYC claim of Bob is revoked, Bob must not be able to receive tokens.
    function test_revokedClaimBlocksReceiving() public {
        _mint(a1, AMT);
        uint256 bBefore = token.balanceOf(b);
        uint256 a1Before = token.balanceOf(a1);

        _revokeClaim(idB);

        vm.prank(a1);
        vm.expectRevert();
        token.transfer(b, AMT);

        assertEq(token.balanceOf(b), bBefore);
        assertEq(token.balanceOf(a1), a1Before);
    }

    // Normal case: KYC-holding investors can still receive mint and transfers.
    function test_verifiedInvestorsStillWork() public {
        uint256 a1Before = token.balanceOf(a1);
        uint256 bBefore = token.balanceOf(b);

        _mint(a1, AMT);
        assertEq(token.balanceOf(a1), a1Before + AMT);

        _transfer(a1, b, AMT);
        assertEq(token.balanceOf(a1), a1Before);
        assertEq(token.balanceOf(b), bBefore + AMT);
    }
}
```
